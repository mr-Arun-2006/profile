version: '3.9'

services:
  postgres:
    image: postgres:16
    container_name: ai-forge-postgres
    restart: unless-stopped
    environment:
      POSTGRES_DB: ai_forge
      POSTGRES_USER: postgres
      POSTGRES_PASSWORD: postgres
    ports:
      - "5432:5432"
    volumes:
      - postgres_data:/var/lib/postgresql/data

  backend:
    image: python:3.12-slim
    working_dir: /app
    volumes:
      - ./backend:/app
    command: bash -lc "pip install -r requirements.txt && uvicorn app.main:app --host 0.0.0.0 --port 8000 --reload"
    environment:
      DATABASE_URL: postgresql+psycopg2://postgres:postgres@postgres:5432/ai_forge
      APP_ENV: development
    ports:
      - "8000:8000"
    depends_on:
      - postgres

  frontend:
    image: node:20-alpine
    working_dir: /app
    volumes:
      - ./frontend:/app
    command: sh -lc "npm install && npm run dev -- --host 0.0.0.0"
    ports:
      - "5173:5173"
    depends_on:
      - backend

volumes:
  postgres_data:
